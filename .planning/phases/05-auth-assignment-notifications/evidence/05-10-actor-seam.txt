import asyncio

from httpx import ASGITransport, AsyncClient

from taskmanager.infrastructure.config.settings import Settings
from taskmanager.main import create_app

app = create_app(
    Settings(
        _env_file=None,
        database_url="postgresql+psycopg://user:pass@localhost:5432/nowhere",
        jwt_secret="a" * 32,
    )
)


async def main() -> None:
    transport = ASGITransport(app=app)
    async with AsyncClient(transport=transport, base_url="http://test") as client:
        for headers, label in (
            (None, "no Authorization header"),
            ({"Authorization": "Basic zzz"}, "Authorization: Basic zzz"),
            ({"Authorization": "Bearer "}, "Authorization: Bearer <empty>"),
            ({"Authorization": "Bearer not.a.real.token"}, "a junk bearer token"),
        ):
            response = await client.get("/api/v1/task-lists", headers=headers)
            print(f"--- {label}")
            print(f"HTTP/1.1 {response.status_code} {response.reason_phrase}")
            print(f"content-type: {response.headers['content-type']}")
            print(f"WWW-Authenticate: {response.headers['WWW-Authenticate']}")
            print(response.text)
            print()


asyncio.run(main())
