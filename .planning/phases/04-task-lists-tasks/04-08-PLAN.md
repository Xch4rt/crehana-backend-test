```python
@router.post("", status_code=status.HTTP_201_CREATED, response_model=TaskListResponse)
async def create_task_list(...) -> TaskListResponse:
    result = await CreateTaskList(uow, clock).execute(payload.to_command(actor_id=actor_id))
    response.headers["Location"] = str(request.url_for("get_task_list", list_id=result.id))

@router.delete("/{list_id}", status_code=status.HTTP_204_NO_CONTENT, response_class=Response)
```

```python
def _raised_name(node: ast.Raise) -> str | None:
    exc = node.exc
    if exc is None:
        return None
    if isinstance(exc, ast.Call):
        exc = exc.func
    if isinstance(exc, ast.Name):
        return exc.id
    if isinstance(exc, ast.Attribute):
```
